01

Flask 基础（2-3 小时，核心）

项目就需要这几个东西，不用学全。

from flask import Flask, jsonify, request, send_file
import sqlite3

app = Flask(__name__)

# 1) 返回 JSON 的 GET 接口 ← 文档要求的核心能力
@app.route('/api/devices', methods=['GET'])
def get_devices():
    conn = sqlite3.connect('netguard.db')
    rows = conn.execute('SELECT * FROM devices').fetchall()
    conn.close()
    return jsonify([{'id': r[0], 'name': r[1], 'ip': r[2]} for r in rows])

# 2) 带参数的接口
@app.route('/api/metrics/<int:device_id>', methods=['GET'])
def get_metrics(device_id):
    limit = request.args.get('limit', 100, type=int)   # 查询参数
    conn = sqlite3.connect('netguard.db')
    rows = conn.execute(
        'SELECT ts, in_octets, out_octets FROM metrics WHERE device_id=? ORDER BY ts DESC LIMIT ?',
        (device_id, limit)
    ).fetchall()
    conn.close()
    return jsonify(rows)

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000, debug=True)

02

SQLite 数据库设计（1-1.5 小时）

重点是「表结构设计」——这是你要讲的板块。建议 3 张表。

-- 1) 设备表
CREATE TABLE IF NOT EXISTS devices (
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    name        TEXT NOT NULL,
    ip          TEXT NOT NULL UNIQUE,
    community   TEXT DEFAULT 'public',
    created_at  TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 2) 指标数据表（核心，采样点都存这里）
CREATE TABLE IF NOT EXISTS metrics (
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    device_id   INTEGER NOT NULL,
    ts          TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    in_octets   INTEGER DEFAULT 0,      -- 累计入站字节
    out_octets  INTEGER DEFAULT 0,      -- 累计出站字节
    in_rate     REAL DEFAULT 0,         -- 算好的实时速率 (bps)
    out_rate    REAL DEFAULT 0,
    FOREIGN KEY (device_id) REFERENCES devices(id)
);
CREATE INDEX IF NOT EXISTS idx_metrics_ts ON metrics(device_id, ts);

-- 3) 告警记录表
CREATE TABLE IF NOT EXISTS alerts (
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    device_id   INTEGER NOT NULL,
    ts          TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    level       TEXT,                   -- info / warning / critical
    metric      TEXT,                   -- 哪个指标触发的
    baseline    REAL,                   -- 当时的基线值
    actual      REAL,                   -- 实际值
    deviation   REAL,                   -- 偏离度
    message     TEXT,
    FOREIGN KEY (device_id) REFERENCES devices(id)
);

03

常用操作 & 必须注意的坑

SQLite 常用操作

import sqlite3

conn = sqlite3.connect('netguard.db')
conn.row_factory = sqlite3.Row          # ← 让结果能按列名访问，强烈建议加
cur = conn.cursor()

# 插入
cur.execute('INSERT INTO metrics (device_id, in_octets, in_rate) VALUES (?, ?, ?)',
            (1, 1234567, 98.5))
conn.commit()

# 查询最近的 100 条
rows = cur.execute('SELECT * FROM metrics WHERE device_id=? ORDER BY ts DESC LIMIT 100', (1,)).fetchall()

# 聚合查询（看板可能要用）
stat = cur.execute('SELECT MAX(in_rate), AVG(in_rate) FROM metrics WHERE device_id=?', (1,)).fetchone()

conn.close()
  • 为什么用 SQLite 而不是 MySQL？（零配置、单文件、够用、赛场不用装服务）
  • 为什么 metrics 表要建索引？（看板要按时间倒序查询，没索引数据一多就卡）
  • 累计计数器（in_octets）和实时速率（in_rate）为什么都要存？
  • 采样间隔 5 秒，一天大概产生多少条数据？（86400/5 = 17280 条/设备/天）
  • 数据量大了怎么办？（定期归档旧数据 / 只保留最近 N 天）

🚨 SQLite 多线程问题

Flask 默认多线程处理请求，而 sqlite3 连接默认不能跨线程用。

# 方案1：每个请求新建连接（简单，推荐）
def get_db():
    conn = sqlite3.connect('netguard.db')
    conn.row_factory = sqlite3.Row
    return conn

# 方案2：连接时加 check_same_thread=False（要自己保证线程安全）
conn = sqlite3.connect('netguard.db', check_same_thread=False)
如果采集线程和后端同时写数据库会报 database is locked。应对：
· 开启 WAL 模式：conn.execute('PRAGMA journal_mode=WAL') ← 推荐，能大幅减少锁冲突
· 设置超时：sqlite3.connect('netguard.db', timeout=10)

CSV 导出功能（0.5 小时）

这是文档要求的产出物之一。

import csv, io
from flask import Response

@app.route('/api/export/<int:device_id>')
def export_csv(device_id):
    conn = sqlite3.connect('netguard.db')
    rows = conn.execute(
        'SELECT ts, in_rate, out_rate FROM metrics WHERE device_id=? ORDER BY ts',
        (device_id,)
    ).fetchall()
    conn.close()

    # 方式1：内存生成 CSV 返回（不用落盘）
    output = io.StringIO()
    writer = csv.writer(output)
    writer.writerow(['时间', '入站速率', '出站速率'])
    writer.writerows(rows)
    output.seek(0)

    return Response(
        '\ufeff' + output.getvalue(),        # \ufeff 是 BOM，防止 Excel 打开中文乱码
        mimetype='text/csv',
        headers={'Content-Disposition': f'attachment; filename=device_{device_id}.csv'}
    )
